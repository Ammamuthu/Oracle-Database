       NAME,
       DB_UNIQUE_NAME,
       DATABASE_ROLE,
       OPEN_MODE,
       LOG_MODE,
       FORCE_LOGGING,
       FLASHBACK_ON

PROMPT 2. INSTANCE INFORMATION
PROMPT ============================================================================

SELECT
       INSTANCE_NAME,
       HOST_NAME,
       VERSION,
       STATUS,
       DATABASE_STATUS,
       STARTUP_TIME
FROM V$INSTANCE;

PROMPT ============================================================================
PROMPT 3. ORACLE VERSION
PROMPT ============================================================================

SELECT BANNER_FULL
FROM V$VERSION;

PROMPT
PROMPT ============================================================================
PROMPT 4. DATABASE COMPONENTS
PROMPT ============================================================================

SELECT
       COMP_NAME,
       VERSION,
       STATUS
FROM DBA_REGISTRY
ORDER BY COMP_NAME;

PROMPT
PROMPT ============================================================================
PROMPT 5. SQL PATCH REGISTRY
PROMPT ============================================================================

SELECT
       PATCH_ID,
       PATCH_TYPE,
       DESCRIPTION,
       ACTION,
       STATUS,
       ACTION_TIME
FROM DBA_REGISTRY_SQLPATCH
ORDER BY ACTION_TIME;

PROMPT
PROMPT ============================================================================
PROMPT 6. INVALID OBJECTS
PROMPT ============================================================================

SELECT
       OWNER,
       OBJECT_TYPE,
       COUNT(*) TOTAL
FROM DBA_OBJECTS
WHERE STATUS='INVALID'
GROUP BY OWNER,OBJECT_TYPE
ORDER BY OWNER,OBJECT_TYPE;

PROMPT
PROMPT ============================================================================
PROMPT 7. TABLESPACE USAGE
PROMPT ============================================================================

SELECT
       TABLESPACE_NAME,
       USED_SPACE,
       TABLESPACE_SIZE,
       ROUND(USED_PERCENT,2) USED_PERCENT
FROM DBA_TABLESPACE_USAGE_METRICS
ORDER BY USED_PERCENT DESC;

PROMPT
PROMPT ============================================================================
PROMPT 8. DATA GUARD STATISTICS
PROMPT ============================================================================

SELECT
       NAME,
       VALUE,
       UNIT
FROM V$DATAGUARD_STATS;

PROMPT
PROMPT ============================================================================
PROMPT 9. ARCHIVE DESTINATION STATUS
PROMPT ============================================================================

SELECT
       DEST_ID,
       STATUS,
       ERROR,
       DESTINATION
FROM V$ARCHIVE_DEST_STATUS
WHERE STATUS <> 'INACTIVE';

PROMPT
PROMPT ============================================================================
PROMPT 10. MANAGED STANDBY PROCESSES
PROMPT ============================================================================

SELECT
       PROCESS,
       STATUS,
       THREAD#,
       SEQUENCE#
FROM V$MANAGED_STANDBY;

PROMPT
PROMPT ============================================================================
PROMPT 11. ARCHIVE GAP
PROMPT ============================================================================

SELECT *
FROM V$ARCHIVE_GAP;

PROMPT
PROMPT ============================================================================
PROMPT 12. LAST APPLIED ARCHIVE
PROMPT ============================================================================

SELECT
       THREAD#,
       MAX(SEQUENCE#) LAST_APPLIED
FROM V$ARCHIVED_LOG
WHERE APPLIED='YES'
GROUP BY THREAD#;

PROMPT
PROMPT ============================================================================
PROMPT 13. LAST GENERATED ARCHIVE
PROMPT ============================================================================

SELECT
       THREAD#,
       MAX(SEQUENCE#) LAST_GENERATED
FROM V$ARCHIVED_LOG
GROUP BY THREAD#;

PROMPT ============================================================================

SELECT
       STATUS,
       COUNT(*) TOTAL
